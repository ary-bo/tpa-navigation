async function downloadDriveExcel(file,token,headerRow){ const isGoogleSheet = file.mimeType === "application/vnd.google-apps.spreadsheet"; let url; if(isGoogleSheet){ url = "https://www.googleapis.com/drive/v3/files/"+ encodeURIComponent(file.id)+ "/export?mimeType="+ encodeURIComponent( "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" ); }else{ url = "https://www.googleapis.com/drive/v3/files/"+ encodeURIComponent(file.id)+"?alt=media"; } const response=await driveRequest(url,token); const buffer=await response.arrayBuffer(); const workbook=XLSX.read(buffer,{type:"array"}); const sheet=workbook.Sheets[workbook.SheetNames[0]]; return XLSX.utils.sheet_to_json(sheet,{ range:headerRow, defval:"" }); }
